        // Номер поточної картки

        let currentCard = 0;


        // Відкриваємо флешкартки

        function openFlashcards() {

            document.getElementById("home-page").classList.add("hidden");

            document.getElementById("flashcards-section").classList.remove("hidden");

            showCard();

        }


        // Повертаємося на головну

        function closeFlashcards() {

            document.getElementById("home-page").classList.remove("hidden");

            document.getElementById("flashcards-section").classList.add("hidden");

        }


        // Показуємо поточну картку

        function showCard() {

            const card = flashcards[currentCard];

            document.getElementById("flashcard-date").textContent = card.date;

            document.getElementById("flashcard-event").textContent = card.event;

            document.getElementById("card-counter").textContent =
                `Картка ${currentCard + 1} із ${flashcards.length}`;

            // Повертаємо картку на передню сторону

            document.getElementById("flashcard-inner").classList.remove("flipped");

        }


        // Перевертаємо картку

        function flipCard() {

            document.getElementById("flashcard-inner").classList.toggle("flipped");

        }


        // Наступна картка

        function nextCard() {

            currentCard++;

            // Після останньої картки повертаємося до першої

            if (currentCard >= flashcards.length) {
                currentCard = 0;
            }

            showCard();

        }


        // Попередня картка

        function previousCard() {

            currentCard--;

            // Перед першою карткою показуємо останню

            if (currentCard < 0) {
                currentCard = flashcards.length - 1;
            }

            showCard();

        }
